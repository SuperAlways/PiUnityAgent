## Global Constraints

- 代码只在 `.worktrees/s3-v3-deviation`（分支 `feature/s3-v3-deviation`）改；主 `main` 只留文档。
- 开工前先把当前未提交的 SOP/audit/spec/plan 文档显式提交到 `main`（**禁止 `git add -A`**），再从 main 建 worktree。
- `PiUnityAgent.Ai` 保持 `netstandard2.1`；禁止 .NET 9+ API。
- `Ai` 零 `UnityEngine` / `UnityLlmTransport` / `GameData` / `Harmony`；不得在 `StreamOptions` 加 `HttpClient`。
- Auth 以 pi `auth/types.ts` / pi.net `Auth/*` 为准；C 类只允许行为后置，不允许形状裁剪。
- Auth 解析以 **stored credential owns provider** 为准；stored 类型无匹配返回 null，不回落 ambient；显式 override ApiKey 走 provider `Resolve` 并带 Env。
- `CredentialStore.ModifyAsync` 的 `null` = 不变（不是删除）；按 provider `SemaphoreSlim` 串行化，禁止 `lock` 内阻塞异步回调。
- `IProvider.Auth` 非空；Faux/测试 provider 必须提供 keyless auth；`ProviderAuth` 至少一个 ApiKey/OAuth。
- 不用 DIM；可选能力用独立接口 / nullable method。
- `ProviderStreams.StreamSimple` 必须用 `SimpleStreamFn`；`ProviderImages/Classifier` 必须用 `ImagesOptions/ClassifierOptions`，不得用 `StreamOptions` 顶替。
- `Identity` 用 `ClientIdentityMerger.Merge` 逐字段合并；`ClientIdentity` 采用 pi 的 `UserAgentMode/ClientAttribution/Version` 形状。
- `LazyStreamObserver` 按 pi：`Push`→`OnEvent`、`End`→`OnResult`；setup 失败 `OnResult(errorMessage)`。
- 所有后置行为保留 Dn；`python scripts/check-deferred.py --root . --gate all` 必须 OK。
- 每任务 TDD：失败测试 → 实现 → 聚焦测试 → 全量 → 提交；禁止 push/merge 无用户批准。
- **最终必须由新的独立 subagent**做偏离审计，结论 PASS。

```csharp
namespace PiUnityAgent.Ai.Providers;

public enum ModelsErrorCode
{
    ModelSource,
    ModelValidation,
    Provider,
    Stream,
    Auth,
    OAuth
}

public sealed class ModelsError : Exception
{
    public ModelsError(ModelsErrorCode code, string message, Exception? cause = null)
        : base(WithCauseDetail(message, cause), cause)
    {
        Code = code;
    }

    public ModelsErrorCode Code { get; }

    private static string WithCauseDetail(string message, Exception? cause)
    {
        if (cause is null) return message;
        var detail = cause.Message.Trim();
        if (detail.Length == 0 || message.Contains(detail, StringComparison.Ordinal)) return message;
        return $"{message}: {detail}";
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text.Json.Nodes;
using System.Threading;
using System.Threading.Tasks;

namespace PiUnityAgent.Ai.Providers.Auth;

public abstract record Credential;

public sealed record ApiKeyCredential(string? Key = null) : Credential
{
    public IReadOnlyDictionary<string, string>? Env { get; init; }
}

public sealed record OAuthCredential : Credential
{
    public required string RefreshToken { get; init; }
    public required string AccessToken { get; init; }
    public required long ExpiresAt { get; init; }
    public IReadOnlyDictionary<string, JsonNode?>? Extra { get; init; }
}

public sealed record CredentialInfo(string ProviderId, AuthType Type);

public interface CredentialStore
{
    Task<Credential?> ReadAsync(string providerId, AuthOperationOptions? options = null);
    Task<IReadOnlyList<CredentialInfo>> ListAsync(AuthOperationOptions? options = null);
    Task<Credential?> ModifyAsync(
        string providerId,
        Func<Credential?, Task<Credential?>> modify,
        AuthOperationOptions? options = null);
    Task DeleteAsync(string providerId, AuthOperationOptions? options = null);
}

public sealed class InMemoryCredentialStore : CredentialStore
{
    private readonly object _gate = new();
    private readonly Dictionary<string, Credential> _credentials = new(StringComparer.Ordinal);
    private readonly Dictionary<string, SemaphoreSlim> _locks = new(StringComparer.Ordinal);

    private SemaphoreSlim LockFor(string providerId)
    {
        lock (_gate)
        {
            if (!_locks.TryGetValue(providerId, out var gate))
            {
                gate = new SemaphoreSlim(1, 1);
                _locks[providerId] = gate;
            }
            return gate;
        }
    }

    public Task<Credential?> ReadAsync(string providerId, AuthOperationOptions? options = null)
    {
        ThrowIfCancelled(options);
        lock (_gate)
        {
            return Task.FromResult<Credential?>(
                _credentials.TryGetValue(providerId, out var credential) ? credential : null);
        }
    }

    public Task<IReadOnlyList<CredentialInfo>> ListAsync(AuthOperationOptions? options = null)
    {
        ThrowIfCancelled(options);
        lock (_gate)
        {
            return Task.FromResult<IReadOnlyList<CredentialInfo>>(
                _credentials.Select(entry => new CredentialInfo(
                    entry.Key,
                    entry.Value is OAuthCredential ? AuthType.OAuth : AuthType.ApiKey)).ToList());
        }
    }

    public async Task<Credential?> ModifyAsync(
        string providerId,
        Func<Credential?, Task<Credential?>> modify,
        AuthOperationOptions? options = null)
    {
        ThrowIfCancelled(options);
        if (modify is null) throw new ArgumentNullException(nameof(modify));

        var gate = LockFor(providerId);
        await gate.WaitAsync(options?.CancellationToken ?? default).ConfigureAwait(false);
        try
        {
            ThrowIfCancelled(options);
            Credential? current;
            lock (_gate) _credentials.TryGetValue(providerId, out current);

            var next = await modify(current).ConfigureAwait(false);
            ThrowIfCancelled(options);
            if (next is not null)
            {
                lock (_gate) _credentials[providerId] = next;
                return next;
            }

            return current;
        }
        finally
        {
            gate.Release();
        }
    }

    public async Task DeleteAsync(string providerId, AuthOperationOptions? options = null)
    {
        ThrowIfCancelled(options);
        var gate = LockFor(providerId);
        await gate.WaitAsync(options?.CancellationToken ?? default).ConfigureAwait(false);
        try
        {
            ThrowIfCancelled(options);
            lock (_gate) _credentials.Remove(providerId);
        }
        finally
        {
            gate.Release();
        }
    }

    private static void ThrowIfCancelled(AuthOperationOptions? options)
        => options?.CancellationToken.ThrowIfCancellationRequested();
}
```

```csharp
private async Task<AuthResult?> ResolveAuthAsync(
    IProvider provider,
    AuthResolutionOverrides? overrides)
{
    var ct = overrides?.Signal ?? default;
    ct.ThrowIfCancellationRequested();
    var auth = provider.Auth;
    if (auth is null) return null;

    // 1. 显式 apiKey 只在 provider 支持 api-key 时走 provider Resolve；合成凭据带 Env。
    if (!string.IsNullOrEmpty(overrides?.ApiKey) && auth.ApiKey is not null)
    {
        var synthetic = new ApiKeyCredential(overrides.ApiKey) { Env = overrides.Env };
        return await ResolveApiKeyAsync(provider, synthetic, ct, overrides.Env).ConfigureAwait(false);
    }

    // 2. stored credential owns provider。
    var stored = await ReadCredentialAsync(provider.Id, ct).ConfigureAwait(false);
    if (stored is OAuthCredential oauth)
    {
        if (auth.OAuth?.ToAuth is null) return null;
        try
        {
            var modelAuth = await auth.OAuth.ToAuth(oauth, ct).ConfigureAwait(false);
            return new AuthResult(modelAuth, null, "OAuth"); // OAuth Extra→env 随 D1/S5
        }
        catch (OperationCanceledException)
        {
            throw;
        }
        catch (Exception error)
        {
            throw new ModelsError(ModelsErrorCode.OAuth,
                $"OAuth auth derivation failed for provider {provider.Id}", error);
        }
    }

    if (stored is ApiKeyCredential apiKey)
    {
        if (auth.ApiKey is null) return null;
        var env = MergeEnv(apiKey.Env, overrides?.Env);
        var credential = env is null ? apiKey : apiKey with { Env = env };
        return await ResolveApiKeyAsync(provider, credential, ct, overrides?.Env).ConfigureAwait(false);
    }

    // stored 类型无匹配：不回落 ambient。
    if (stored is not null) return null;
    if (auth.ApiKey is null) return null;

    // 3. 仅无 stored 时走 ambient。
    return await ResolveApiKeyAsync(provider, null, ct, overrides?.Env).ConfigureAwait(false);
}

private async Task<AuthResult?> ResolveApiKeyAsync(
    IProvider provider,
    ApiKeyCredential? credential,
    CancellationToken ct,

    catch (Exception error)
```

```csharp
private static IReadOnlyDictionary<string, string?>? MergeHeaders(
    IReadOnlyDictionary<string, string?>? provider,
    IReadOnlyDictionary<string, string?>? request)
{
    if (provider is null && request is null) return null;
    var merged = new Dictionary<string, string?>(StringComparer.Ordinal);
    if (provider is not null)
        foreach (var entry in provider) merged[entry.Key] = entry.Value;
    if (request is not null)
    {
        foreach (var entry in request)
        {
            foreach (var existing in merged.Keys.ToList())
            {
                if (string.Equals(existing, entry.Key, StringComparison.OrdinalIgnoreCase))
                    merged.Remove(existing);
            }
            merged[entry.Key] = entry.Value;
        }
    }
    return merged;
}

private static IReadOnlyDictionary<string, string>? MergeEnv(
    IReadOnlyDictionary<string, string>? provider,
    IReadOnlyDictionary<string, string>? request)
{
    if (provider is null && request is null) return null;
    var merged = new Dictionary<string, string>(provider ?? new Dictionary<string, string>());
    if (request is not null) foreach (var entry in request) merged[entry.Key] = entry.Value;
    return merged;
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using PiUnityAgent.Ai.Types;

namespace PiUnityAgent.Ai.Providers;

public static class ModelOperations
{
    private static readonly ModelThinkingLevel[] ExtendedThinkingLevels =
    {
        ModelThinkingLevel.Off,
        ModelThinkingLevel.Minimal,
        ModelThinkingLevel.Low,
        ModelThinkingLevel.Medium,
        ModelThinkingLevel.High,
        ModelThinkingLevel.XHigh,
        ModelThinkingLevel.Max,
    };

    public static ModelType GetModelType(ProviderModel model) => model.Type;

    public static bool IsModelType(ProviderModel model, ModelType type) => model.Type == type;

    public static bool HasApi(Model model, string api)
        => model.Type == ModelType.Chat && model.Api == api;

    public static bool ModelsAreEqual(ProviderModel? a, ProviderModel? b)
        => a is not null && b is not null
           && a.Type == b.Type && a.Id == b.Id && a.Provider == b.Provider;

    public static UsageCost CalculateCost(ProviderModel model, Usage usage)
    {
        var inputTokens = usage.Input + usage.CacheRead + usage.CacheWrite;
        var rates = model.Cost;
        var matchedThreshold = -1;
        foreach (var tier in model.Cost.Tiers ?? Array.Empty<ModelCostTier>())
        {
            if (inputTokens > tier.InputTokensAbove && tier.InputTokensAbove > matchedThreshold)
            {
                rates = new ModelCost(tier.Input, tier.Output, tier.CacheRead, tier.CacheWrite);
                matchedThreshold = tier.InputTokensAbove;
            }
        }

        var longWrite = usage.CacheWrite1h ?? 0;
        var shortWrite = usage.CacheWrite - longWrite;
        var input = rates.Input / 1_000_000 * usage.Input;
        var output = rates.Output / 1_000_000 * usage.Output;
        var cacheRead = rates.CacheRead / 1_000_000 * usage.CacheRead;
        var cacheWrite = (rates.CacheWrite * shortWrite + rates.Input * 2 * longWrite) / 1_000_000;
        return new UsageCost
        {
            Input = input,
            Output = output,
            CacheRead = cacheRead,
            CacheWrite = cacheWrite,
            Total = input + output + cacheRead + cacheWrite,
            TotalFromProvider = false,
        };
    }

    public static IReadOnlyList<ModelThinkingLevel> GetSupportedThinkingLevels(Model model)
    {
        if (!model.Reasoning) return new[] { ModelThinkingLevel.Off };

        return ExtendedThinkingLevels
            .Where(level =>
            {
                var hasKey = model.ThinkingLevelMap?.ContainsKey(level) == true;
                var mapped = hasKey ? model.ThinkingLevelMap![level] : null;
                if (hasKey && mapped is null) return false;
                if (level is ModelThinkingLevel.XHigh or ModelThinkingLevel.Max)
                {
                    return hasKey && mapped is not null;
                }
                return true;
            })
            .ToList();
    }

    public static ModelThinkingLevel ClampThinkingLevel(Model model, ModelThinkingLevel level)
    {
        var available = GetSupportedThinkingLevels(model);
        if (available.Contains(level)) return level;

        var requestedIndex = Array.IndexOf(ExtendedThinkingLevels, level);
        if (requestedIndex < 0) return available[0];

        for (var i = requestedIndex; i < ExtendedThinkingLevels.Length; i++)
        {
            if (available.Contains(ExtendedThinkingLevels[i])) return ExtendedThinkingLevels[i];
        }
        for (var i = requestedIndex - 1; i >= 0; i--)
        {
            if (available.Contains(ExtendedThinkingLevels[i])) return ExtendedThinkingLevels[i];
        }
        return available[0];
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using PiUnityAgent.Ai.Events;
using PiUnityAgent.Ai.Providers.Auth;
using PiUnityAgent.Ai.Types;

namespace PiUnityAgent.Ai.Providers;

public static class ProviderFactory
{
    public static IProvider Create(CreateProviderOptions options)
    {
        if (options is null) throw new ArgumentNullException(nameof(options));
        if (options.Auth is null || (options.Auth.ApiKey is null && options.Auth.OAuth is null))
            throw new ModelsError(ModelsErrorCode.Provider,
                $"Provider {options.Id}: at least one of auth.apiKey/auth.oauth is required.");
        if (options.Api is not null && options.ApiByApi is not null)
            throw new ModelsError(ModelsErrorCode.Provider,
                $"Provider {options.Id}: specify Api or ApiByApi, not both.");

        var streams = options.Api is not null
            ? new[] { options.Api }
            : (options.ApiByApi?.Values ?? Array.Empty<ProviderStreams>()).ToArray();
        var hasImages = options.Images?.Count > 0;
        var hasClassifiers = options.Classifiers?.Count > 0;
        if (streams.Length == 0 && !hasImages && !hasClassifiers)
            throw new ModelsError(ModelsErrorCode.Provider,
                $"Provider {options.Id}: at least one of Api/ApiByApi/Images/Classifiers is required.");

        return new CreatedProvider(options, options.Api, options.ApiByApi);
    }

    private sealed class CreatedProvider : IProvider, IFilterableProvider, IImagesProvider, IClassifierProvider
    {
        private readonly CreateProviderOptions _options;
        private readonly ProviderStreams? _single;
        private readonly IReadOnlyDictionary<string, ProviderStreams>? _byApi;

        public CreatedProvider(
            CreateProviderOptions options,
            ProviderStreams? single,
            IReadOnlyDictionary<string, ProviderStreams>? byApi)
        {
            _options = options;
            _single = single;
            _byApi = byApi;
        }

        public string Id => _options.Id;
        public string Name => _options.Name ?? _options.Id;
        public string? BaseUrl => _options.BaseUrl;
        public IReadOnlyDictionary<string, string?>? Headers => _options.Headers;
        public ProviderAuth Auth => _options.Auth;

        public IReadOnlyList<Model> GetModels() => _options.Models.OfType<Model>().ToList();
        public IReadOnlyList<ProviderModel> GetAllModels() => _options.Models;

        public IReadOnlyList<Model> FilterModels(IReadOnlyList<Model> models, Credential? credential)
            => _options.FilterModels?.Invoke(models, credential) ?? models;

        public IReadOnlyList<ProviderModel>? FilterAllModels(
            IReadOnlyList<ProviderModel> models, Credential? credential)
            => _options.FilterAllModels?.Invoke(models, credential);

        public IAssistantMessageEventStream Stream(
            Model model, TranscriptContext context, StreamOptions? options)
        {
            var streams = ApiFor(model);
            return streams is null ? MissingApi(model) : streams.Stream(model, context, options);
        }

        public IAssistantMessageEventStream StreamSimple(
            Model model, TranscriptContext context, SimpleStreamOptions? options)
        {
            var streams = ApiFor(model);
            return streams is null ? MissingApi(model) : streams.StreamSimple(model, context, options);
        }

        private ProviderStreams? ApiFor(Model model) => _single ?? _byApi?.GetValueOrDefault(model.Api);

        private IAssistantMessageEventStream MissingApi(Model model)
            => LazyStream.Run(model, () => throw new ModelsError(
                ModelsErrorCode.Stream,
                $"Provider {_options.Id} has no API implementation for \"{model.Api}\""));

        // D4/D5：S3 只保证能力接口与 api-map 分派骨架；真实图片/分类行为在 2B。
        public Task<AssistantImages> GenerateImagesAsync(
            ImageModel model, ImagesContext context, ImagesOptions? options)
            => _options.Images is not null && _options.Images.TryGetValue(model.Api, out var implementation)
                ? implementation.GenerateImages(model, context, options)
                : throw new NotSupportedException("Deferred to 2B: image generation (D4)");

        public Task<ClassifierResult> ClassifyAsync(
            ClassifierModel model, ClassifierContext context, ClassifierOptions? options)
            => _options.Classifiers is not null && _options.Classifiers.TryGetValue(model.Api, out var implementation)
                ? implementation.Classify(model, context, options)
                : throw new NotSupportedException("Deferred to 2B: classification (D5)");
    }
```

```csharp
using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;
using PiUnityAgent.Ai.Types;

namespace PiUnityAgent.Ai.Providers.Auth;

public sealed record ModelsStoreEntry
{
    public required IReadOnlyList<ProviderModel> Models { get; init; }
    public long? LastModified { get; init; }
    public long? CheckedAt { get; init; }
    public string? Etag { get; init; }
}

public sealed record ModelsStoreOperationOptions
{
    public CancellationToken CancellationToken { get; init; }
}

public interface IModelsStore
{
    Task<ModelsStoreEntry?> ReadAsync(string providerId, ModelsStoreOperationOptions? options = null);
    Task WriteAsync(string providerId, ModelsStoreEntry entry, ModelsStoreOperationOptions? options = null);
    Task DeleteAsync(string providerId, ModelsStoreOperationOptions? options = null);
}

public sealed class InMemoryModelsStore : IModelsStore
{
    private readonly object _gate = new();
    private readonly Dictionary<string, ModelsStoreEntry> _entries = new(StringComparer.Ordinal);

    public Task<ModelsStoreEntry?> ReadAsync(string providerId, ModelsStoreOperationOptions? options = null)
    {
        ThrowIfCancelled(options);
        lock (_gate)
        {
            return Task.FromResult(_entries.TryGetValue(providerId, out var entry) ? entry : null);
        }
    }

    public Task WriteAsync(string providerId, ModelsStoreEntry entry, ModelsStoreOperationOptions? options = null)
    {
        ThrowIfCancelled(options);
        if (entry is null) throw new ArgumentNullException(nameof(entry));
        lock (_gate) _entries[providerId] = entry;
        return Task.CompletedTask;
    }

    public Task DeleteAsync(string providerId, ModelsStoreOperationOptions? options = null)
    {
        ThrowIfCancelled(options);
        lock (_gate) _entries.Remove(providerId);
        return Task.CompletedTask;
    }

    private static void ThrowIfCancelled(ModelsStoreOperationOptions? options)
        => options?.CancellationToken.ThrowIfCancellationRequested();
}

public sealed record ModelsPublication
{
    public ModelsStoreEntry? Persist { get; init; }
    public bool Delete { get; init; }
    public Action? Update { get; init; }
}

    public Credential? Credential { get; init; }

    public string? BaseUrl { get; init; }
    public required Func<ModelsPublication, Task<bool>> Publish { get; init; }
    public bool AllowNetwork { get; init; }
    public bool Force { get; init; }
    public CancellationToken CancellationToken { get; init; }
}
```
