```csharp
using System.Text;
using PiUnityAgent.Ai.Types;

namespace PiUnityAgent.Ai.Tests.Fakes;

public sealed class MockFetch
{
    private readonly object _gate = new();
    private readonly Queue<Func<FetchRequest, FetchOptions?, CancellationToken, Task<IFetchResponse>>> _queue = new();
    private int _callCount;

    public List<FetchRequest> Requests { get; } = new();
    public int CallCount => Volatile.Read(ref _callCount);
    public FetchFunction Fetch => FetchAsync;

    public void Enqueue(Func<FetchRequest, FetchOptions?, CancellationToken, Task<IFetchResponse>> handler)
    {
        lock (_gate) _queue.Enqueue(handler);
    }

    public void EnqueueText(string text, int status = 200, IReadOnlyDictionary<string, string?>? headers = null)
    {
        var responseHeaders = new FetchHeaders();
        if (headers is not null)
        {
            foreach (var entry in headers) responseHeaders[entry.Key] = entry.Value;
        }

        EnqueueBytes(status, Encoding.UTF8.GetBytes(text), responseHeaders);
    }

    public void EnqueueBytes(int status, byte[] body, FetchHeaders? headers = null)
        => Enqueue((_, _, _) => Task.FromResult<IFetchResponse>(new MockFetchResponse(status, body, headers)));

    public void EnqueueException(FetchException exception)
        => Enqueue((_, _, _) => Task.FromException<IFetchResponse>(exception));

    private async Task<IFetchResponse> FetchAsync(
        FetchRequest request,
        FetchOptions? options,
        CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();

        Func<FetchRequest, FetchOptions?, CancellationToken, Task<IFetchResponse>> handler;
        lock (_gate)
        {
            if (_queue.Count == 0)
                throw new InvalidOperationException("No mock responses queued.");
            handler = _queue.Dequeue();
            Requests.Add(request);
            Interlocked.Increment(ref _callCount);
        }

        return await handler(request, options, cancellationToken).ConfigureAwait(false);
    }
}

public sealed class MockFetchResponse : IFetchResponse
{
    private readonly MemoryStream _body;
    private int _disposed;
    private int _abortCount;

    public MockFetchResponse(int statusCode, byte[] body, FetchHeaders? headers = null)
    {
        StatusCode = statusCode;
        Headers = headers ?? new FetchHeaders();
        _body = new MemoryStream(body, writable: false);
    }

    public int StatusCode { get; }
    public FetchHeaders Headers { get; }
    public Stream Body => _body;
    public int AbortCount => Volatile.Read(ref _abortCount);

    public void Abort()
    {
        if (Interlocked.Exchange(ref _disposed, 1) == 0)
        {
            Interlocked.Increment(ref _abortCount);
            _body.Dispose();
        }
    }

    public ValueTask DisposeAsync()
    {
        Abort();
        return default;
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using System.Text.Json;
using System.Text.Json.Serialization;
using System.Threading;
using System.Threading.Tasks;

namespace PiUnityAgent.Ai.Types;

/// <summary>pi `fetch` 的 C# 等价物；由宿主注入（B 方案）。</summary>
public delegate Task<IFetchResponse> FetchFunction(
    FetchRequest request,
    FetchOptions? options,
    CancellationToken cancellationToken);

public sealed record FetchRequest(
    string Method,
    string Url,
    FetchHeaders Headers,
    ReadOnlyMemory<byte> Body)
{
    /// <summary>大请求（图片/多模态）用工厂重建流；设置后忽略 Body。必须可重放。</summary>
    public Func<CancellationToken, Task<Stream>>? BodyFactory { get; init; }
}

/// <summary>Headers：名称大小写不敏感；请求中 value=null 表示“抑制默认头”。</summary>
public sealed class FetchHeaders : Dictionary<string, string?>
{
    public FetchHeaders() : base(StringComparer.OrdinalIgnoreCase) { }

    public FetchHeaders(IDictionary<string, string?> values)
        : base(values, StringComparer.OrdinalIgnoreCase) { }
}

public sealed record FetchOptions
{
    public TransportKind? Kind { get; init; }
    public TimeSpan? ConnectTimeout { get; init; }
    public TimeSpan? FirstByteTimeout { get; init; }
    public TimeSpan? IdleTimeout { get; init; }
    public int? MaxResponseBytes { get; init; }
    public IReadOnlyDictionary<string, string>? Env { get; init; }
    public string? SessionId { get; init; }
}

public interface IFetchResponse : IAsyncDisposable
{
    int StatusCode { get; }
    FetchHeaders Headers { get; }

    /// <summary>响应体；任意状态码都可读（错误体也是 Body）。</summary>
    Stream Body { get; }

    /// <summary>立即中止底层连接；幂等。</summary>
    void Abort();
}

public enum FetchErrorKind
{
    Connect,
    Tls,
    Proxy,
    Protocol,
    FirstByteTimeout,
    IdleTimeout,
    Cancelled,
    Aborted,
    Overflow,
    Unsupported
}

public sealed class FetchException : Exception
{
    public FetchException(FetchErrorKind kind, string message, Exception? innerException = null)
        : base(message, innerException)
    {
        Kind = kind;
    }

    public FetchErrorKind Kind { get; }
    public int? StatusCode { get; init; }
    public FetchHeaders? Headers { get; init; }
    public string? ErrorBody { get; init; }
    public bool IsRetryable { get; init; }
}

[JsonConverter(typeof(TransportKindJsonConverter))]
public enum TransportKind
{
    Auto,
    Sse,
    WebSocket,
    WebSocketCached
}

public sealed class TransportKindJsonConverter : JsonConverter<TransportKind>
{
    public override TransportKind Read(
        ref Utf8JsonReader reader,
        Type typeToConvert,
        JsonSerializerOptions options)
        => reader.GetString() switch
        {
            "auto" => TransportKind.Auto,
            "sse" => TransportKind.Sse,
            "websocket" => TransportKind.WebSocket,
            "websocket-cached" => TransportKind.WebSocketCached,
            var value => throw new JsonException($"Unknown transport kind: {value}")
        };

    public override void Write(
        Utf8JsonWriter writer,
        TransportKind value,
        JsonSerializerOptions options)
        => writer.WriteStringValue(value switch
        {
            TransportKind.Auto => "auto",
            TransportKind.Sse => "sse",
            TransportKind.WebSocket => "websocket",
            TransportKind.WebSocketCached => "websocket-cached",
            _ => throw new JsonException($"Unknown transport kind: {value}")
        });
}
```

```csharp
public sealed class InMemoryCredentialStore : CredentialStore

    private readonly Dictionary<string, Credential> _credentials = new(StringComparer.Ordinal);

    public Task<Credential?> GetAsync(string providerId, CancellationToken ct = default)
    {
        ct.ThrowIfCancellationRequested();

        {
            return Task.FromResult<Credential?>(
                _credentials.TryGetValue(providerId, out var credential) ? credential : null);
        }
    }

        return Task.CompletedTask;
    }

    {
        ct.ThrowIfCancellationRequested();
        lock (_gate) _credentials.Remove(providerId);
        return Task.CompletedTask;
    }
```

```csharp
using System;

namespace PiUnityAgent.Ai.Types;

public static class TranscriptNormalization
{
    /// <summary>
    /// 把 Context.systemPrompt 与 Context.tools 折成前导 SystemMessage；
    /// 两者都为空时不插入 system 消息。原消息顺序不变。
    /// </summary>
    public static TranscriptContext NormalizeContext(Context context)
    {
        if (context is null) throw new ArgumentNullException(nameof(context));

        var hasPrompt = !string.IsNullOrEmpty(context.SystemPrompt);
        var hasTools = context.Tools is { Count: > 0 };
        if (!hasPrompt && !hasTools)
        {
            return new TranscriptContext { Messages = context.Messages };
        }

        var system = new SystemMessage
        {
            Timestamp = 0,
            Content = MessageContent.From(context.SystemPrompt ?? string.Empty),
            ToolsAdded = hasTools ? context.Tools : null,
        };

        var messages = new Message[context.Messages.Count + 1];
        messages[0] = system;
        for (var index = 0; index < context.Messages.Count; index++)
        {
            messages[index + 1] = context.Messages[index];
        }

        return new TranscriptContext { Messages = messages };
    }
}
```

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using PiUnityAgent.Ai.Types;

namespace PiUnityAgent.Ai.Events;

public static class LazyStream
{
    /// <summary>
    /// 同步返回 outer 流；返回时立即调度 factory（Task.Run）。factory 只启动一次。
    /// setup 失败（含 OCE）统一为 Error(Error)；Aborted 由 provider 内层流负责。
    /// 对应 pi api/lazy.ts 与 pi.net Api/Lazy.cs。
    /// </summary>
    public static IAssistantMessageEventStream Run(
        Model model,
        Func<Task<IAssistantMessageEventStream>> factory,
        TelemetrySpan? telemetry = null)
    {
        if (model is null) throw new ArgumentNullException(nameof(model));
        if (factory is null) throw new ArgumentNullException(nameof(factory));

        var startedAt = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds();
        var outer = new AssistantMessageEventStream();

        // fire-and-forget：与 pi.net Api/Lazy.cs 一致，setup 在返回后异步调度；
        // ForwardAsync 内部捕获所有异常并以终态事件收场。
        _ = Task.Run(() => ForwardAsync(outer, model, factory, startedAt));
        return outer;
    }

    private static async Task ForwardAsync(
        AssistantMessageEventStream outer,
        Model model,
        Func<Task<IAssistantMessageEventStream>> factory,
        long startedAt)
    {
        try
        {
            var inner = await factory().ConfigureAwait(false)
                ?? throw new InvalidOperationException("LazyStream factory returned null.");

            await foreach (var @event in inner.ConfigureAwait(false))
            {
                outer.Push(@event);
            }

            var final = await inner.WaitForDoneAsync().ConfigureAwait(false);
            outer.End(final);
        }
        catch (Exception error)
        {
            var message = new AssistantMessage
            {
                Content = Array.Empty<ContentBlock>(),
                Api = model.Api,
                Provider = model.Provider,
                Model = model.Id,
                Usage = new Usage(),
                StopReason = StopReason.Error,
                ErrorMessage = error.Message,
                Timestamp = startedAt,
            };
            outer.Push(new AssistantMessageEvent.Error(StopReason.Error, message));
            outer.End(message);
        }
    }
}
```

```csharp
namespace PiUnityAgent.Ai.Tests.Providers;

internal static class ProviderTestSupport
{
    public static Model CreateModel(string id = "m1", string provider = "p1")
        => new()
        {
            Id = id,
            Name = id,
            Api = "faux",
            Provider = provider,
            BaseUrl = "http://localhost:0",
        };

    public static IAssistantMessageEventStream DoneStream(
        Model model,
        TranscriptContext context,
        StreamOptions? options)
    {
        var stream = new AssistantMessageEventStream();
        var message = new AssistantMessage
        {
            Content = Array.Empty<ContentBlock>(),
            Api = model.Api,
            Provider = model.Provider,
            Model = model.Id,
            Usage = new Usage(),
            StopReason = StopReason.Stop,
            Timestamp = 1,
        };
        stream.Push(new AssistantMessageEvent.Done(StopReason.Stop, message));
        stream.End(message);
        return stream;
    }

    public static StaticProvider CreateProvider(
        string id = "p1",
        StreamFn? stream = null,
        ProviderAuth? auth = null,
        IReadOnlyList<Model>? models = null,
        string? baseUrl = null,
        IReadOnlyDictionary<string, string?>? headers = null)
        => new(
            id,
            id,
            models ?? new[] { CreateModel("m1", id) },
            stream ?? DoneStream,
            baseUrl ?? "http://localhost:0",
            headers,
            auth);

    public static async Task<List<AssistantMessageEvent>> CollectAsync(
        IAssistantMessageEventStream stream,
        CancellationToken cancellationToken = default)
    {
        var events = new List<AssistantMessageEvent>();
        await foreach (var @event in stream.WithCancellation(cancellationToken))
        {
            events.Add(@event);
        }
        return events;
    }
}
```

```csharp
        Context context,
        SimpleStreamOptions? options = null)
    {
        var transcript = TranscriptNormalization.NormalizeContext(context);
        return LazyStream.Run(model, async () =>
        {
            var provider = RequireProvider(model.Provider);
            var key = await ResolveApiKeyAsync(provider, options?.ApiKey).ConfigureAwait(false);
            if (key is not null)
```

```csharp
    {
        var provider = GetProvider(providerId);
        if (provider is null) return null;
        if (provider.Auth is null) return new AuthCheck(true);

        var auth = await GetAuthAsync(providerId).ConfigureAwait(false);
        return auth?.Credential switch
        {
            ApiKeyCredential => new AuthCheck(true, AuthType.ApiKey),
            OAuthCredential => new AuthCheck(true, AuthType.OAuth),
            _ => null,
        };

        AuthOperationOptions? options = null)

            var credential = (await GetAuthAsync(entry.Id).ConfigureAwait(false))?.Credential;
            var models = SafeChatModels(entry);
            if (entry is IRefreshableProvider refreshable)
            {
                models = refreshable
                    .FilterModels(models.Cast<ProviderModel>().ToList(), credential)
                    .OfType<Model>()
                    .ToList();
            }

        return result;

            var models = SafeAllModels(entry);
            if (entry is IRefreshableProvider refreshable)
            {
                models = refreshable.FilterModels(models, credential);
            }

            result.AddRange(models.Where(model => model.Type == type));
        }

        return result;
    }

        foreach (var entry in SelectProviders(provider))

            var models = SafeAllModels(entry);
            if (entry is IRefreshableProvider refreshable)

                models = refreshable.FilterModels(models, credential);

        AuthOperationOptions? options = null)

        if (provider.Auth?.OAuth?.Logout is not null)

        await _credentials.DeleteAsync(providerId).ConfigureAwait(false);

    private async Task<AuthResult?> ResolveAuthAsync(
        IProvider provider,
        AuthResolutionOverrides? overrides)
    {
        if (overrides?.Credential is not null)
        {
            return new AuthResult(overrides.Credential, "options");
        }

        if (!string.IsNullOrEmpty(overrides?.ApiKey))
        {
            return new AuthResult(new ApiKeyCredential(overrides.ApiKey), "options");
        }

        var stored = await _credentials.GetAsync(provider.Id).ConfigureAwait(false);
        if (stored is ApiKeyCredential apiKey && !string.IsNullOrEmpty(apiKey.Key))
        {
            return new AuthResult(apiKey, "store");
        }

        if (stored is OAuthCredential oauth)
        {
            // TODO(S5, D1): OAuth 刷新/派生请求认证。
            return new AuthResult(oauth, "store");
        }

        var resolve = provider.Auth?.ApiKey?.Resolve;
        if (resolve is not null)
        {
            var key = await resolve(_authContext, null, default).ConfigureAwait(false);
            if (!string.IsNullOrEmpty(key))
            {
                return new AuthResult(new ApiKeyCredential(key), "provider");
            }
        }

        return null;
    }

    private IReadOnlyList<IProvider> SelectProviders(string? providerId)
        => providerId is null
            ? GetProviders()
            : GetProvider(providerId) is { } provider
                ? new[] { provider }
                : Array.Empty<IProvider>();

    private async Task<bool> IsAvailableAsync(
        IProvider provider,
        AuthOperationOptions? options)
    {
        if (provider.Auth is null) return true;
        var check = await CheckAuthAsync(provider.Id, options).ConfigureAwait(false);
        return check?.Configured == true;
    }
```
