```csharp
    private static IReadOnlyDictionary<string, JsonNode?>? ReadExtra(string providerId, JsonObject entry)
    {
        var extra = new Dictionary<string, JsonNode?>(StringComparer.Ordinal);

        // 旧版把未知字段嵌在 "extra"：读入时并进平铺结果，写出时不再产生该键。
        if (entry.TryGetPropertyValue("extra", out var legacyNode) && legacyNode is not null)
        {
            if (legacyNode is not JsonObject legacyObject)
                throw new ModelsError(
                    ModelsErrorCode.Auth, $"Credential entry '{providerId}' has a non-object 'extra'");
            foreach (var pair in legacyObject) extra[pair.Key] = pair.Value?.DeepClone();
        }

        foreach (var pair in entry)
        {
            if (pair.Key == "extra" || IsOAuthReservedKey(pair.Key)) continue;
            extra[pair.Key] = pair.Value?.DeepClone();
        }

        return extra.Count == 0 ? null : extra;
    }

    private static bool IsOAuthReservedKey(string key)
        => key is "type" or "refresh" or "access" or "expires";
```

    private static StreamOptions ApplyAuthOptions(Model model, StreamOptions? options, AuthResult auth)
    {
        options ??= new StreamOptions();
        var sessionHeaders = ClientIdentityHelpers.GetOpenCodeSessionHeaders(
            new ClientIdentityTarget(model.Provider, model.BaseUrl), options.SessionId) is { } session
            ? session.ToDictionary(entry => entry.Key, entry => (string?)entry.Value)
            : null;
        // pi models.ts:883：session 头最底层，auth.headers 次之，请求 headers 最高。
        var headers = MergeHeaders(sessionHeaders, MergeHeaders(auth.Auth.Headers, options.Headers));
        var env = MergeEnv(auth.Env, options.Env);
        return options with
        {
            ApiKey = options.ApiKey is null ? auth.Auth.ApiKey : options.ApiKey,
            Headers = headers,
            Env = env,
        };
    }

    private static SimpleStreamOptions ApplyAuthOptions(Model model, SimpleStreamOptions? options, AuthResult auth)
